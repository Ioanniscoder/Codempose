## 🎵 Example Workflows

### Create a Simple Melody
```python
"""My First Study"""

MELODY_LILY = r"""
\relative c'' {
    \time 4/4
    \key c \major
    c4 d4 e4 f4 |
    g2 a2 |
    b4 a4 g4 f4 |
    e2 d2 |
    c1
}
""".strip()

def build_score_data():
    from lilypond_parser import parse_lilypond_to_data
    melody_data = parse_lilypond_to_data(MELODY_LILY)
    return {
        'metadata': {'title': 'My First Study'},
        'parts': {'Melody': melody_data['parts']['Melody']}
    }

if __name__ == '__main__':
    from project_template import run_pipeline_from_file
    run_pipeline_from_file(__file__)
```

### Add Harmonization
```python
from harmonic_engine import harmonize_melody
from music_data import data_to_part, part_to_data

# Parse melody
melody_data = parse_lilypond_to_data(MELODY_LILY)
melody_events = melody_data['parts']['Melody']

# Convert to music21 Part
melody_part = data_to_part(melody_events)

# Harmonize with progression
score = harmonize_melody(melody_part, "I - IV - V - I", "C")

# Extract bass line
bass_events = part_to_data(score.parts[1])

# Build final score
return {
    'metadata': {'title': 'Harmonized Study'},
    'parts': {
        'Melody': melody_events,
        'Bass': bass_events
    }
}
```

### Use music21 Transformations
```python
from music21 import interval

# Parse melody
melody_part = data_to_part(melody_events)

# Transpose up a major third
transposed = melody_part.transpose('M3')

# Convert back to canonical format
transposed_events = part_to_data(transposed)
```
