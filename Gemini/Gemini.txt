def parse_rhythms(rhythm_str, expected_count):

    Parse rhythm string into list of rhythm codes.

    

    Formats:

    - Single rhythm: "w" -> apply to all

    - Multiple: "h-q-q-h" -> one per event

    """

    rhythm_map = {

        'h': 2.0,    # half

        'q': 1.0,    # quarter

        'e': 0.5,    # eighth

        's': 0.25,   # sixteenth

        'h.': 3.0,   # dotted half

        'q.': 1.5,   # dotted quarter
